                                              # 向量

                                              struct Vector
{
    int x,y;
    Vector(int a,int b):x(0),y(0){}
    Vector(int a,int b):x(a),y(b){}
    Vector operator+(Vector a){
        return Vector(x+a.x,y+a.y);
    }
    Vector operator-(Vector a){
        return Vector(x-a.x,y-a.y);
    }
};

                                              # 点积

                                              # 定义

                                              A⋅B=∣A∣∣B∣cos⁡θA\cdot B = |A| |B| \cos\thetaA⋅B=∣A∣∣B∣cosθ

                                              template <typename T>
T Dot(Vector p,Vector q){
    return p.x*q.x+p.y*q.y;
}

                                              # 应用

                                              1. 判断AAA与BBB的夹角是钝角还是锐角

                                                点积有正负：

                                                • 若Dot(A,B)>0Dot(A, B) > 0Dot(A,B)>0，AAA与BBB的夹角为锐角；
                                                • 若Dot(A,B)<0Dot(A, B) < 0Dot(A,B)<0，AAA与BBB的夹角为钝角；
                                                • 若Dot(A,B)=0Dot(A, B) = 0Dot(A,B)=0，AAA与BBB的夹角为直角。

                                              2. 求向量AAA的长度

                                              double Len(Vector A){
	return sqrt(Dot(A,A));
}
                                              1. 求向量AAA与BBB的夹角大小
                                              double Angle(Vector A,Vector B){
	return acos(Dot(A,B)/Len(A)/Len(B));
}

                                              # 叉积

                                              # 定义

                                              A×B=∣A∣∣B∣sin⁡θA\times B = |A| |B| \sin\thetaA×B=∣A∣∣B∣sinθ

                                              template <typename T>
T Cross(Vector p,Vector q){
    return p.x*q.y+p.y*q.x;
}
                                              • 两个向量的叉积是一个带正负号的数值。A×BA\times BA×B的几何意义为向量AAA和BBB形成的平行四边形的“有向”面积，这个面积有正负。

                                              # 应用

                                              1. 判断向量AAA、BBB的方向关系

                                                • 若A×B>0A\times B > 0A×B>0，BBB在AAA的逆时针方向；
                                                • 若A×B<0A\times B < 0A×B<0，BBB在AAA的顺时针方向；
                                                • 若A×B=0A\times B = 0A×B=0，BBB与AAA共线，可能是同方向的，也可能是反方向的。

                                              2. 计算两向量构成的平行四边形有向面积

                                                • 三个点AAA、BBB、CCC，以AAA为公共点，得到222个向量B−AB-AB−A和C−AC-AC−A，它们构成的平行四边形，面积是：

                                              double Area2(Point A,Point B,Point C){ return Cross(B-A, C-A); } ```

                                              1. 计算三点构成的三角形的面积

                                                • 三个点AAA、BBB、CCC构成的三角形面积，等于平行四边形面积Area2(A,B,C)Area2(A, B, C)Area2(A,B,C)的二分之一。

                                              2. 向量旋转

                                                • 使向量(x, y)绕起点逆时针旋转，设旋转角度为θ，那么旋转后的向量(x’, y’)是：

                                                x’=xcos⁡θ–ysin⁡θy’=xsin⁡θ+ycos⁡θ x’ = x\cos\theta – y\sin\theta\\y’ = x\sin\theta + y\cos\theta x’=xcosθ–ysinθy’=xsinθ+ycosθ

                                                Vector Rotate(Vector A, double rad){  
	return Vector(A.x*cos(rad)-A.y*sin(rad),A.x*sin(rad)+A.y*cos(rad));
}

                                              # 直线的表示

                                              • 用直线上的两个点来表示。
                                              • ax+by+c=0ax + by + c = 0ax+by+c=0，普通式。
                                              • y=kx+by = kx + by=kx+b，斜截式。
                                              • P=P0+vtP = P_0 + vtP=P0​+vt，点向式。P0(x0,y0)P_0 (x_0, y_0)P0​(x0​,y0​)是直线上的一个点；vvv是方向向量。

                                              # 点和直线的位置关系

                                              点在直线左侧、在右侧、在直线上。用直线上的两点p1p_1p1​和p2p_2p2​，与点ppp构成两个向量，用叉积的正负判断方向。

                                              int Point_line_relation(Point p, Line v){
	int c = sgn(Cross(p-v.p1,v.p2-v.p1));
	if(c < 0)return 1;       //1：p在v的左边
	if(c > 0)return 2;       //2：p在v的右边
	return 0;                  //0：p在v上
}

                                              # 点到直线的距离

                                              • 已知点ppp和直线v(p1,p2)v(p_1,p_2)v(p1​,p2​)，求ppp到vvv的距离。
                                              • 首先用叉积求ppp、p1p_1p1​、p2p_2p2​构成的平行四边形面积，然后用面积除以平行四边形的底边长，也就是线段(p1,p2)(p_1, p_2)(p1​,p2​)的长度，就得到了平行四边形的高，即ppp点到直线的距离。
                                              double Dis_point_line(Point p, Line v){
	return fabs(Cross(p-v.p1,v.p2-v.p1))/Distance(v.p1,v.p2);
}

                                              # 点在直线上的投影

                                              已知直线上两点p1p_1p1​、p2p_2p2​、以及直线外一点ppp，求投影点p0p_0p0​。

                                              Point Point_line_proj(Point p, Line v){
    double k=Dot(v.p2-v.p1,p-v.p1)/Len2(v.p2-v.p1);
    return v.p1+(v.p2-v.p1)*k;
}

                                              # 点关于直线的对称点

                                              求一个点ppp对一条直线vvv的镜像点。先求点ppp在直线上的投影qqq，再求对称点p’p’p’。

                                              Point Point_line_symmetry(Point p, Line v){
    Point q = Point_line_proj(p,v);
    return Point(2*q.x-p.x,2*q.y-p.y);
}

                                              # 点到线段的距离

                                              点ppp到线段ABABAB的距离。在以下333个距离中取最小值：

                                              • 从ppp出发对ABABAB做垂线，如果交点在ABABAB线段上，这个距离就是最小值；
                                              • ppp到AAA的距离；
                                              • ppp到BBB的距离。
                                              double Dis_point_seg(Point p, Segment v){
	if(sgn(Dot(p- v.p1,v.p2-v.p1))<0 || sgn(Dot(p- v.p2,v.p1-v.p2))<0)
		return min(Distance(p,v.p1),Distance(p,v.p2));
    return Dis_point_line(p,v); //点的投影在线段上
}

                                              # 两条直线的位置关系

                                              int Line_relation(Line v1, Line v2){
    if (sgn(Cross(v1.p2 - v1.p1, v2.p2 - v2.p1)) == 0)
    {
        if (Point_line_relation(v1.p1, v2) == 0)
            return 1; // 1 重合
        else
            return 0; // 0 平行
    }
    return 2; // 2 相交
}

                                              # 两条直线的交点

                                              444个点A、B、C、DA、B、C、DA、B、C、D，组成两条直线ABABAB和CDCDCD，交点是PPP

                                              Point Cross_point(Point a,Point b,Point c,Point d){ //Line1:ab,  Line2:cd
    double s1 = Cross(b-a,c-a);
    double s2 = Cross(b-a,d-a);            //叉积有正负
    return Point(c.x*s2-d.x*s1,c.y*s2-d.y*s1)/(s2-s1);
}

                                              # 两个线段是否相交

                                              如果一条线段的两端在另一条线段的两侧，那么两个端点与另一线段的产生的两个叉积正负相反，也就是说两个叉积相乘为负。如果两条线段互相满足这一点，那么就是相交的。

                                              bool Cross_segment(Point a, Point b, Point c, Point d)
{
    // Line1:ab,  Line2:cd
    double c1 = Cross(b - a, c - a), c2 = Cross(b - a, d - a);
    double d1 = Cross(d - c, a - c), d2 = Cross(d - c, b - c);
    return sgn(c1) * sgn(c2) < 0 && sgn(d1) * sgn(d2) < 0;
    // 1相交；0不相交
}

                                              # 多边形

                                              # 判断点在多边形内部

                                              • 000角法：把点PPP和多边形的每个点连接，逐个计算角度，绕多边形一周，看多边形相对于这个点总共转了多少度。如果是360360360度，说明点在多边形内；如果是000度，说明点在多边形外；如果是180180180度，说明点在多边形边界上。

                                              # 求多边形面积

                                              double Polygon_area(Point *p, int n){ // Point *p表示多边形
    double area = 0;
    for (int i = 0; i < n; i++)
        area += Cross(p[i], p[(i + 1) % n]);
    return area / 2; //面积有正负
}
