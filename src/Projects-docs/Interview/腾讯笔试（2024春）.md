小红定义一个点是“好点”，当且仅当这个点的所有邻边都是红边。

现在请你求出这个无向图“好点”的数量。

注：如果一个节点没有任何邻边，那么它也是好点。

# 2

### 题意

小红拿到了一个链表。她准备将这个链表断裂成两个链表，再拼接到一起，使得链表从头节点到尾部升序。你能帮小红判断能否达成目的吗？

给定的为一个链表数组，你需要对于数组中每个链表进行一次“是”或者“否”的答案回答，并返回布尔数组。

每个链表的长度不小于 2，且每个链表中不包含两个相等的元素。所有链表的长度之和保证不超过10^5

### 解法

解法不难。

难点：

-   给的为链表和类，要求实现类（我开始写半天读入）
-   题目意思看清楚，是将链表分为两段，然后再拼接，不是从中选出数变成两个链表再拼接（别问我为啥知道）

### Code

```cpp
class Solution {
public:
    /**
     * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
     *
     * 
     * @param lists ListNode类vector 
     * @return bool布尔型vector
     */



    vector<bool> canSorted(vector<ListNode*>& lists) {
        vector<bool>res;
        auto check=[&](vector<int>&v){
            int flag=0;

            int idx=1;
            for(;idx<v.size();idx++){
                if(v[idx]<v[idx-1])break;
            }

            if(idx==v.size())return true;

            vector<int>vv;
            for (int i = idx; i < v.size(); i++)
            {
                vv.push_back(v[i]);
            }
            for (int i = 0; i < idx; i++)
            {
                vv.push_back(v[i]);
            }
            
            
            for (int i = 1; i < vv.size(); i++)
            {
                if(vv[i]<vv[i-1])return false;
            }

            return true;
        };
        for(ListNode *head:lists){
            vector<int>v;
            for(;head!=nullptr;head=head->next){
                v.push_back(head->val);
            }
            res.push_back(check(v));
        }

        return res;
        
    }
};
```

### 题意

小红拿到了一个有 n 个节点的无向图，这个图初始并不是连通图。

现在小红想知道，添加恰好一条边使得这个图连通，有多少种不同的加边方案？

### 解法

并查集板子题

### Code

```cpp
#include<bits/stdc++.h>
#define ll long long
#define ptn putchar('\n')
using namespace std;

inline ll read() {ll x = 0, z = 1;char c = getchar();while (!isdigit(c)) {if (c == '-')z = -1;c = getchar();}while (isdigit(c)) {x = (x << 1) + (x << 3) + (c ^ 48);c = getchar();}return z * x;}
template<typename T>
inline void writ(T x,char c=-1){if(x<0) {putchar('-');x=(~x)+1;}if(x>9)writ(x/10);putchar(x-x/10*10+48);if(~c)putchar(c);}



void ClearDewy(){
    int n=read(),m=read();

    vector<int>f(n+1);
    iota(f.begin(),f.end(),0);

    function<int(int)>myfind=[&](int x){
        return x==f[x]?x:f[x]=myfind(f[x]);
    };
    auto uni=[&](int x,int y){
        if(myfind(x)==myfind(y))
            return;
        f[myfind(y)]=myfind(x);
    };

    int x,y;
    for (int i = 1; i <= m; i++)
    {
        x=read();y=read();
        uni(x,y);
    }
    map<int,int>mp;
    for (int i = 1; i <= n; i++)
    {
        mp[myfind(i)]++;
    }
    if(mp.size()>2){
        writ(0);
    }else if(mp.size()==2){
        writ(1LL*mp.begin()->second*mp.rbegin()->second);
    }else{
        writ(1LL*n*(n-1)/2);
    }
}
```

## 4

### 题意

小红拿到了一个数组，她准备将数组分割成 k 段，使得每段内部做按位异或后，再全部求和。小红希望最终这个和尽可能大，你能帮帮她吗？

$1\leq k\leq n\leq 400$

### 解法

$dp$，设 $dp[i][j]$ 为前 $i$ 个数分成 $j$ 段的最大异或和。则
$$
dp[i][j]=max_{p=1}^{i-1} dp[p][j-1]+(pre[i]\oplus  pre[p])
$$
其中 $pre[i]$ 为前 $i$ 个数异或和

### Code

```cpp
#include<bits/stdc++.h>
#define ll long long
#define ptn putchar('\n')
using namespace std;

inline ll read() {ll x = 0, z = 1;char c = getchar();while (!isdigit(c)) {if (c == '-')z = -1;c = getchar();}while (isdigit(c)) {x = (x << 1) + (x << 3) + (c ^ 48);c = getchar();}return z * x;}
template<typename T>
inline void writ(T x,char c=-1){if(x<0) {putchar('-');x=(~x)+1;}if(x>9)writ(x/10);putchar(x-x/10*10+48);if(~c)putchar(c);}


void ClearDewy(){
    int n=read(),k=read();

    vector<ll>pre(n+1);
    vector<vector<ll>>dp(n+1,vector<ll>(k+1));

    for (int i = 1; i <= n; i++)
    {
        pre[i]=pre[i-1]^read();
    }

    for (int i = 1; i <= n; i++)
    {
        dp[i][1]=pre[i];
        for (int j = 2; j <= k&&j<=i; j++)
        {
            for (int p = j-1; p < i; p++)
            {
                dp[i][j]=max(dp[i][j],dp[p][j-1]+(pre[i]^pre[p]));
            }
        }
    }

    writ(dp[n][k]);
}



int main(){
    //cin.tie(nullptr)->sync_with_stdio(false);

    //int T=read();while (T--)
    ClearDewy();
    return 0;
}
```

## 5

### 题意

小红拿到了一个字符矩阵，她可以从任意一个地方出发，希望走 6 步后恰好形成"tencent"字符串。小红想知道，共有多少种不同的行走方案？

注：每一步可以选择上、下、左、右中任意一个方向进行行走。不可行走到矩阵外部。

### 解法

设 $dp[i][j][k]$ 为第 $k$ 步走到 $(i,j)$ 点，枚举 $k$ 转移即可。

### Code

```cpp
#include<bits/stdc++.h>
#define ll long long
#define ptn putchar('\n')
using namespace std;

inline ll read() {ll x = 0, z = 1;char c = getchar();while (!isdigit(c)) {if (c == '-')z = -1;c = getchar();}while (isdigit(c)) {x = (x << 1) + (x << 3) + (c ^ 48);c = getchar();}return z * x;}
template<typename T>
inline void writ(T x,char c=-1){if(x<0) {putchar('-');x=(~x)+1;}if(x>9)writ(x/10);putchar(x-x/10*10+48);if(~c)putchar(c);}



const int N=1005;
char M[N][N];

ll dp[N][N][7];

const string s="tencent";

int dir[4][2]={
    {1,0},{-1,0},{0,1},{0,-1}
};

void ClearDewy(){
    int m=read(),n=read();
    for (int i = 1; i <= m; i++)
    {
        cin>>M[i]+1;
    }

    for (int i = 1; i <= m; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            if(M[i][j]=='t'){
                dp[i][j][0]=1;
            }
        }
    }
    
    for (int k = 1; k < 7; k++)
    {
        for (int i = 1; i <= m; i++)
        {
            for (int j = 1; j <= n; j++)
            {
                if(M[i][j]!=s[k])
                    continue;

                for (int d = 0; d < 4; d++)
                {
                    dp[i][j][k]+=dp[i+dir[d][0]][j+dir[d][1]][k-1];
                }
            }
        }
    }
    
    ll res=0;

    for (int i = 1; i <= m; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            res+=dp[i][j][6];
        }
    }
    writ(res);
}



int main(){
    //cin.tie(nullptr)->sync_with_stdio(false);

    //int T=read();while (T--)
    ClearDewy();
    return 0;
}
```
