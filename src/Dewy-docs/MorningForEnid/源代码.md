## morning.py

```python
from datetime import date, datetime
from wechatpy import WeChatClient
from wechatpy.client.api import WeChatMessage
import urllib,random,json,schedule,time,requests

class to_girlfriend():
    def __init__(self):
        self.today = datetime.now()
        with open('data.json', 'r', encoding='UTF-8') as json_data:
            data = json.load(json_data)
            self.start_date = data["start_date"]
            self.city = data["city"]
            self.birthday = data["birthday"]
            self.app_id = data["app_id"]
            self.app_secret = data["app_secret"]
            self.user_id = data["user_id"]
            self.template_id = data["template_id"]
            self.constellation=data["constellation"]
            self.owner_id=data["owner_id"]
            self.constellation_template_id=data["constellation_template_id"]
            self.birthday_template_id=data["birthday_template_id"]

    def get_weather(self):
        url = "http://apis.juhe.cn/simpleWeather/query"
        head = {
            "Content-Type": "application/x-www-form-urlencoded",
        }
        data = {
            "city": self.city,
            "key": "434f6793ec0b70a8366875532707b766"
        }
        res = requests.post(url=url, headers=head, data=data).json()
        realtime = res["result"]["realtime"]
        future = res["result"]["future"]

        # [place]天气【[week]~[month]/[days]】：[now_weather]，[now_temperature]℃，[now_wind_direction]（[now_wind_power]）
        weather_now = "当前天气：{info}；温度：{temperature}℃；风向：{direct}；风力：{power}；空气质量：{aqi}\n".format(
            info=realtime["info"], temperature=realtime["temperature"], direct=realtime["direct"],
            power=realtime["power"], aqi=realtime["aqi"])
        weather_future = ''
        for i in range(4):
            weather_future = weather_future + "{date}天气：{weather}；温度：{temperature}；风向：{direct}\n".format(
                date=future[i]["date"], weather=future[i]["weather"], temperature=future[i]["temperature"],
                direct=future[i]["direct"])
        return weather_now, weather_future


    def get_count(self):
        delta = self.today - datetime.strptime(self.start_date, "%Y-%m-%d")
        return delta.days+1

    def get_birthday(self):
        next = datetime.strptime(str(date.today().year) + "-" + self.birthday, "%Y-%m-%d")
        if next < datetime.now():
            next = next.replace(year=next.year + 1)
        if next.month==datetime.now().month and next.day==datetime.now().day:
            return "今天是坤宝生日！！祝坤宝生日快乐！！！"
        return ((next - self.today).days+1).__str__()

    def get_words(self):
        words = requests.get("https://api.shadiao.pro/chp")
        if words.status_code != 200:
            return self.get_words()
        return words.json()['data']['text']

    def get_random_color(self):
        return "#%06x" % random.randint(0, 0xFFFFFF)

    def get_constellation(self):
        url = "http://web.juhe.cn/constellation/getAll"  # ?consName=摩羯座&type=today&key=7455dea5e5e24c857cf2ca4d4f54b18c
        params = urllib.parse.urlencode({
            'consName': self.constellation,
            'type': "today",
            'key': "7455dea5e5e24c857cf2ca4d4f54b18c",
        })
        req = requests.post(url=url, params=params).json()
        return (req)

    def send_to(self):
        client = WeChatClient(self.app_id, self.app_secret)
        wm = WeChatMessage(client)
        weather_now,weather_future= self.get_weather()
        data = {
                "weather_now": {"value": weather_now, "color": self.get_random_color()},
                "weather_future": {"value": weather_future, "color": self.get_random_color()},
                "love_days": {"value": self.get_count(), "color": self.get_random_color()},
                "birthday_left": {"value": self.get_birthday(), "color": self.get_random_color()},
                "words": {"value": self.get_words(), "color": self.get_random_color()},
                }
        c_data=self.get_constellation()
        constellation_data={
            "constellation": {"value": self.constellation, "color": self.get_random_color()},
            "c_color": {"value": c_data['color'], "color": self.get_random_color()},
            "c_health": {"value": c_data['health'], "color": self.get_random_color()},
            "c_love": {"value": c_data['love'], "color": self.get_random_color()},
            "c_work": {"value": c_data['work'], "color": self.get_random_color()},
            "c_money": {"value": c_data['money'], "color": self.get_random_color()},
            "c_number": {"value": c_data['number'], "color": self.get_random_color()},
            "c_all": {"value": c_data['all'], "color": self.get_random_color()},
            "c_summary": {"value": c_data['summary'], "color": self.get_random_color()},
            }
        if len(self.get_birthday())>4:
            id=self.birthday_template_id
        else:
            id=self.template_id
        res = wm.send_template(self.user_id, id, data)
        wm.send_template(self.user_id, self.constellation_template_id, constellation_data)
        print(res)
        wm.send_template(self.owner_id, id, data)
        wm.send_template(self.owner_id, self.constellation_template_id, constellation_data)



def work():
    msg = to_girlfriend()
    msg.send_to()


def main():
  work()
  schedule.every().day.at("05:20").do(work)
  schedule.every().day.at("13:14").do(work)
  schedule.every().day.at("20:00").do(work)
  schedule.every(20).to(28).hours.do(work)
  while True:
    schedule.run_pending()
    time.sleep(1)

# 按间距中的绿色按钮以运行脚本。
if __name__ == '__main__':
    main()
```
