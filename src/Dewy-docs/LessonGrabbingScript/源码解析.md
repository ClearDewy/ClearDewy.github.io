# 获取 pyfaid 和 jxqdm参数，用来访问进入选课中心，否则选课请求出错
def getPyfailJxqdm(text:str):
    pattern = r"loadPyfas\((.*?)\);"

    match = re.findall(pattern, text, re.DOTALL)
    if match:
        values = [value.strip().strip("'") for value in match[0].split(",")]
        return values[0],values[2]
    else:
        log.error("当前不为选课时间范围")
        exit(0)


def login():
    # 获取验证码
    verify = requests.post(url='http://xk.scuec.edu.cn/xsxk/servlet/ImageServlet')
    verify_png=verify.content
    # f=open('verify.png','wb')
    # f.write(verify_png)
    # f.close()
    verifyCode=get_verifyCode(verify_png)
    # verifyCode=input("输入验证码：")
    if len(verifyCode)!=4:
        return
    log.info("验证码：{}".format(verifyCode))
    # 请求登录
    req=requests.post(url="http://xk.scuec.edu.cn/xsxk/loadData.xk",headers=head,cookies=verify.cookies,data={
        "username":username,
        "password":password,
        "method":"checkLogin",
        "verifyCode":verifyCode
    }).json()
    if not req['success']:
        log.info(req['message'])
        return False
    # 重新加载页面，并从页面中获取pyfaid 和 jxqdm参数
    pyfaild,jxqdm=getPyfailJxqdm(requests.post(url="http://xk.scuec.edu.cn/xsxk/login.xk",headers=head,cookies=verify.cookies,data={
        "username": username,
        "password": password,
        "verifyCode": verifyCode
    }).text)
    # 访问选课中心，否则选课请求出错
    requests.post(
        url="http://xk.scuec.edu.cn/xsxk/xkjs.xk?pyfaid={}&jxqdm={}&data-frameid=main&data-timer=2000&data-proxy=proxy.xk".format(pyfaild,jxqdm),
        headers=head, cookies=verify.cookies)
    log.info("登录成功!")
    # 返回登录的cookie 用来选课
    return verify.cookies

def logout(cooky):
    requests.get(url="http://xk.scuec.edu.cn/xsxk/logout.xk",headers=head,cookies=cooky)

def solve(cooky,calssdata):
    global visi_num
    while True:
        # 发送选课请求
        req = requests.post(url="http://xk.scuec.edu.cn/xsxk/xkOper.xk", headers=head, cookies=cooky, data={
            "method": calssdata["method"],
            "jxbid": calssdata["jxbid"],
            "glJxbid": calssdata["glJxbid"],
        },timeout=1000).json()
        if req['success']:
            log.info("课程ID:{} 选课成功！！".format(calssdata["jxbid"]))
            visi[calssdata["jxbid"]]=True
            visi_num=visi_num+1
            return
        else:
            log.error("选课失败："+req['message'])
            if req['message']=="用户登录已失效请重新登录":
                return
            # req['message'].count("已达到教学班容量")or
            if  req['message'].count("已经选择"):
                visi[calssdata["jxbid"]] = True
                visi_num = visi_num + 1
                return

# 获取登录成功的cookie用于选课
def get_cooky():
    cooky = ''

    # 自行添加cookie，可使脚本和网页同时在线
    # cooky={
    #     "JSESSIONID":"0000vjUfFJaC-g2iO3ArczPdbjF:1almjnh8e"
    # }
    while not cooky:
        cooky = login()
    return cooky

def start_xk(cooky):
    # 进程池，使用多进程使得每个课程同时进行选择
    processes = []
    for i in idList:
        if not visi[i["jxbid"]]:
            xk_process = Process(target=solve, args=(cooky, i))
            xk_process.start()
            processes.append(xk_process)
    for i in processes:
        i.join()

def main():
    pass
    cooky=''
    while visi_num<len(visi):
        cooky = get_cooky()
        start_xk(cooky)

    logout(cooky)

if __name__ == '__main__':
    main()
