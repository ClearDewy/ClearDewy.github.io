# 拓扑排序

```cpp
void topesort()
{
    queue<int> q;
    for (int i = 1; i <= n; i++)
    {
        if (!deg[i])
        {
            q.push(i);
            break;
        }
    }
    int h,to;
    while (!q.empty())
    {
        h = q.front();
        q.pop();
        for (int i = head[h]; i; i = e[i].next)
        {
            to = e[i].to;
            if (!(--deg[to]))
            {
                q.push(to);
            }
        }
    }
}
```
