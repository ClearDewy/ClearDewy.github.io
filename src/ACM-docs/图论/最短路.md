## SPFA

```cpp
const int N=1e5+5;
struct edge
{
    int to,next;ll w;
}e[N<<1];
int head[N],ent=0;
inline void addedge(int x,int y,ll w){
    e[++ent]={y,head[x],w};
    head[x]=ent;
}
int dis[N];
bool spfa(int n, int s) {
  queue<int> q;int x,y;
  vector<int>cnt(n+1);vector<bool>vi(n+1);
  memset(dis, 0X3f, sizeof(dis));
  dis[s] = 0, vi[s] = 1;
  q.push(s);
  while (!q.empty()) {
    x = q.front();
    q.pop(), vi[x] = 0;
    for (int i=head[x];i;i=e[i].next) {
      y=e[i].to;
      if (dis[x] > dis[y] + e[i].w) {
        dis[y] = dis[x] + e[i].w;
        cnt[y] = cnt[x] + 1;  // 记录最短路经过的边数
        if (cnt[y] >= n) return false;
        // 在不经过负环的情况下，最短路至多经过 n - 1 条边
        // 因此如果经过了多于 n 条边，一定说明经过了负环
        if (!vi[y]) q.push(y), vi[y] = 1;
      }
    }
  }
  return true;
}
```
