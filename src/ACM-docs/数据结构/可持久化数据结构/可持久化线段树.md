# 可持久化线段树

## 主席树

主席树全称是可持久化权值线段树

## 引入

先引入一道题目：给定$n$个整数构成的序列$a$，将对于指定的闭区间$[l,r]$查询其区间内的第$k$小值。

你该如何解决？

一种可行的方案是：使用主席树。 主席树的主要思想就是：保存每次插入操作时的历史版本，以便查询区间第$k$小。

## 实现

![persistent-seg](https://raw.githubusercontent.com/ClearDewy/TyporaImg/main/img/202211041800162.png)

如图，在每次更新时新建一条链，每次最多增加$\lceil\log_2n \rceil+1$个节点

-   每次修改后的根节点必不相同
-   在回溯的时候使用一个新的节点，指向当前节点更新后的左孩子和右孩子

用一个数组将每次修改后的根节点编号存下来，然后进行线段树操作即可

#include<bits/stdc++.h>
#define ll long long
#define ptn putchar('\n')

inline ll read() {ll x = 0, z = 1;char c = getchar();while (!isdigit(c)) {if (c == '-')z = -1;c = getchar();}while (isdigit(c)) {x = (x << 1) + (x << 3) + (c ^ 48);c = getchar();}return z * x;}
template<typename T>
inline void writ(T x,char c=-1){if(x<0) {putchar('-');x=(~x)+1;}if(x>9)writ(x/10);putchar(x-x/10*10+48);if(~c)putchar(c);}

const int N=1e5;
struct node
{
    int l,r,sum;
}tr[N<<5];
int trid=0,root[N];

int build(int l,int r){
    if(l==r){
        return ++trid;
    }
    int mid=l+r>>1,id=++trid;;
    tr[id].l=build(l,mid);
    tr[id].r=build(mid+1,r);
    return id;
}

int update(int pre,int l,int r,int idx){
    int id=++trid,mid=l+r>>1;
    tr[id]=tr[pre];tr[id].sum++;
    if(l==r){
        return id;
    }
    if(idx<=mid){
        tr[id].l=update(tr[pre].l,l,mid,idx);
    }else{
        tr[id].r=update(tr[pre].r,mid+1,r,idx);
    }
    return id;
}

int query_k_th(int prel,int prer,int l,int r,int k){
    if(l==r){
        return l;
    }
    int x=tr[tr[prer].l].sum-tr[tr[prel].l].sum,mid=l+r>>1;
    if(k<=x){
        return query_k_th(tr[prel].l,tr[prer].l,l,mid,k);
    }else{
        return query_k_th(tr[prel].r,tr[prer].r,mid+1,r,k-x);
    }
}

int query_lessOrEqual_k(int prel,int prer,int l,int r,int k){
    if(l==r){
        return tr[prer].sum-tr[prel].sum;
    }
    int mid=l+r>>1;
    if(k<=mid){
        return query_lessOrEqual_k(tr[prel].l,tr[prer].l,l,mid,k);
    }else{
        return tr[tr[prer].l].sum-tr[tr[prel].l].sum+query_lessOrEqual_k(tr[prel].r,tr[prer].r,mid+1,r,k);
    }
}

void ClearDewy(){
    int n=read(),m=read();
    vector<int>a(n+1);
    for (int i = 1; i <= n; i++)
    {
        a[i]=read();
    }
    // ...排序+离散化处理 b 
    set<int>st(a.begin()+1,a.end());
    vector<int>b(st.begin(),st.end());
    
    root[0]=build(1,b.size());
    for (int i = 1; i <= n; i++)
    {
        root[i]=update(root[i-1],1,b.size(),lower_bound(b.begin(),b.end(),a[i])-b.begin()+1);
    }

    int l,r,k;
    while (m--)
    {
        l=read();r=read();k=read();
        // [l,r]中第k小的数是多少
        // writ(a[query_k_th(root[l-1],root[r],l,r,k)],'\n');
        // [l,r]中小于等于k的数有多少个

        int idx=upper_bound(b.begin(),b.end(),k)-b.begin();
        writ(idx==0?0:query_lessOrEqual_k(root[l-1],root[r],1,b.size(),idx),'\n');
    }
}
