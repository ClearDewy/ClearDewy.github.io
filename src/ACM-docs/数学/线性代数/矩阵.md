# 矩阵

## 矩阵乘法

```cpp
const int X = 105;
ll mod = 1e9 + 7;
struct martrix
{
    int x=0, y=0;  //行数  列数
    ll M[X][X] = { 0 };
    inline martrix operator*(const martrix& b) const{
        martrix c;
        if (y!=b.x)
        {
            return c;
        }
        c.x = x; c.y = b.y;

        ll r;
        for (int i = 1; i <= x; i++) {
            for (int k = 1; k <= y; k++) {
                r = M[i][k];
                for (int j = 1; j <= c.y; ++j) {
                    c.M[i][j] += b.M[k][j] * r;
                    c.M[i][j] %= mod;
                }

            }
        }
        return c;
    }
};
```

## 矩阵快速幂

```cpp
martrix fp_martrix(martrix x, ll y) {
    martrix base;
    base.x = x.x; base.y = x.y;
    for (int i = 1; i < X; i++)
    {
        base.M[i][i] = 1;
    }
    while (y)
    {
        if (y&1)base = base * x;
        x = x * x;
        y >>= 1;
    }
    return base;
}
```
