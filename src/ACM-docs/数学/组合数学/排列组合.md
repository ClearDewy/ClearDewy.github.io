# 排列组合

## 基本公式
$$
\mathrm A_n^m = n(n-1)(n-2) \cdots (n-m+1) = \frac{n!}{(n - m)!} \\

\binom{n}{m}= \mathrm C_n^m = \frac{\mathrm A_n^m}{m!} = \frac{n!}{m!(n - m)!}
$$

- 选取$x,y$为较小的数，通常在$20$以内
```cpp
inline ll C(int x,int y) {		//y中选x个
    ll base = 1;
    if (x==0)return base;
    for (int i = y; i > y-x; i--)
        base = base * i / (y - i + 1);
    return base;
}
```
- 当模数为质数，且$x,y$不超过$10^8$

```cpp
const int N=1e6+5,mod=1e9+7;
ll p[N],invp[N];

inline ll fp(ll x, ll y) {
    ll base = 1;
    while (y){
        if (y&1)base =base*x%mod;
        x=x*x%mod;y >>= 1;
    }
    return base;
}

void pre_work(){
    p[0]=1;
    for (int i = 1; i < N; i++)
        p[i]=p[i-1]*i%mod;
    invp[N-1]=fp(p[N-1],mod-2);     //求逆元
    for (int i = N-2; ~i ; i--)
        invp[i]=invp[i+1]*i%mod;
}

ll C(int x,int y){
    return p[x]*invp[x-y]%mod*invp[y]%mod;
}
```

-   当模数为质数，且模数较小时

```cpp
inline ll lucas(ll x,ll y){
    if(!y)return 1;
    return C(x%mod,y%mod)*lucas(x%mod,y%mod)%mod;
}
```

## 组合数性质 | 二项式推论

$$
\begin{aligned}
\binom{n}{m}&=\frac{n}{m} \binom{n-1}{m-1}\\
\binom{n}{m}&=\binom{n-1}{m}+\binom{n-1}{m-1}\\
\binom{n}{0}+\binom{n}{1}+\cdots +\binom{n}{n}&=2^n\\
\sum_{i=0}^n(-1)^i\binom{n}{i}&=[n=0]\\
\sum_{i=0}^m\binom{n}{i}\binom{m}{m-i}&=\binom{n+m}{m} \ (n\geq m)\\
\sum_{i=0}^ni\binom{n}{i}&=n2^{n-1}
\end{aligned}
$$
