# 多项式快速幂

## 定义

求解$f(x)=g^k(x)\pmod {x^n}$

## 解法

$$
\begin{align}
f(x)&=g^k(x)\pmod {x^n}\\
&=e^{\ln{g^k(x)}}\pmod {x^n}\\
&=e^{k\ln g(x)}
\end{align}
$$

#include<bits/stdc++.h>
#define ll long long
#define ptn putchar('\n')
using namespace std;

inline ll read() {ll x = 0, z = 1;char c = getchar();while (!isdigit(c)) {if (c == '-')z = -1;c = getchar();}while (isdigit(c)) {x = (x << 1) + (x << 3) + (c ^ 48);c = getchar();}return z * x;}
template<typename T>
inline void writ(T x,const char &c = -1){if(x<0) {putchar('-');x=(~x)+1;}if(x>9)writ(x/10);putchar(x-x/10*10+48);if(~c)putchar(c);}

struct Poly
{
    static const int mod=998244353,g=3,gi=332748118;
    static inline int pg(int x){if((x&-x)==x)return x;x |= x >> 1;x |= x >> 2;x |= x >> 4;x |= x >> 8;x |= x >> 16;return x + 1;}
    static inline int fp(int x, ll y) {int base = 1;while (y){if (y&1)base=(ll)base*x%mod;x=(ll)x*x%mod;y >>= 1;}return base;}
    int num;vector<int>poly,poly_inv,rev,poly_ln,poly_exp;
    int& operator[](int idx){return this->poly[idx];}
    int size(){return poly.size();}
    Poly(int n):num(n),poly(vector<int>(pg(n))){};

    static inline void NTT(vector<int>&poly,vector<int>&rev,bool typ,int size=-1){
        if(size==-1)size=poly.size();
        for (int i = 0; i < size; i++)
            if(i<rev[i])swap(poly[i],poly[rev[i]]);
        ll x,y;
        for (int i = 1; i < size; i<<=1)
        {
            ll gn=fp(typ?g:gi,(mod-1)/(i<<1));
            for (int j = 0; j < size; j+=(i<<1)){
                ll g0=1;
                for (int k = 0; k < i; k++,g0=g0*gn%mod)
                {
                    x = poly[j + k]; y = g0 * poly[i + j + k] % mod;
                    poly[j + k] = (x + y) % mod;
                    poly[i + j + k] = (x - y + mod) % mod;
                }
            }
        }
        if(typ)return;
        ll inv_size=fp(size,mod-2);
        for (int i = 0; i < size; i++)poly[i]=poly[i]*inv_size%mod;
    }

    static inline void Poly_inv(vector<int> &poly,vector<int>&poly_inv,vector<int>&rev,vector<int> &C,int deg){
        if(deg==1){
            poly_inv[0]=fp(poly[0],mod-2);return;
        }
        Poly_inv(poly,poly_inv,rev,C,(deg+1)>>1);
        int size=pg(deg<<1);
        for (int i = 0; i < size; i++){
            rev[i]=(rev[i>>1]>>1)|((i&1)?(size>>1):0);
            C[i]=(i < deg ? poly[i] : 0);
        }
        NTT(poly_inv,rev,1,size);NTT(C,rev,1,size);
        for (int i = 0; i < size; i++)
            poly_inv[i]=(2-(ll)poly_inv[i]*C[i]%mod+mod)%mod*poly_inv[i]%mod;
        NTT(poly_inv,rev,0,size);
        fill(poly_inv.begin()+deg,poly_inv.begin()+size,0);
    }

    static inline void Poly_ln(vector<int> &poly,vector<int>poly_inv,vector<int>&poly_ln,vector<int>&rev,int num,int size){
        for (int i = 0; i < num; i++)
            poly_ln[i]=(ll)poly[i+1]*(i+1)%mod;
        for (int i = 0; i < size; i++)rev[i]=(rev[i>>1]>>1)|((i&1)?(size>>1):0);
        NTT(poly_ln,rev,1);NTT(poly_inv,rev,1);
        for (int i = 0; i < size; i++)poly_ln[i]=(ll)poly_ln[i]*poly_inv[i]%mod;
        NTT(poly_ln,rev,0);
        for (int i = num; i ; i--)
            poly_ln[i]=(ll)poly_ln[i-1]*fp(i,mod-2)%mod;
        poly_ln[0]=0;
    }

    static inline void Poly_exp(vector<int> &poly,vector<int>&rev,vector<int>&poly_exp,int deg){
        if(deg==1){
            poly_exp[0]=1;return;
        }
        Poly_exp(poly,rev,poly_exp,(deg+1)>>1);
        int size=pg(deg)<<1;
        vector<int> poly_inv(size),poly_ln(size),C(size);
        Poly_inv(poly_exp,poly_inv,rev,C,deg);
        Poly_ln(poly_exp,poly_inv,poly_ln,rev,deg,size);
        for (int i = 0; i < deg; i++)
            poly_ln[i]=(poly[i]-poly_ln[i])<0?poly[i]-poly_ln[i]+mod:poly[i]-poly_ln[i];
        size=pg(deg<<1);
        for (int i = 0; i < size; i++)
            rev[i]=(rev[i>>1]>>1)|((i&1)?(size>>1):0);
        NTT(poly_exp,rev,1,size);NTT(poly_ln,rev,1,size);
        for (int i = 0; i < size; i++)
            poly_exp[i]=(ll)poly_exp[i]*(poly_ln[i]+1)%mod;
        NTT(poly_exp,rev,0,size);
        fill(poly_exp.begin()+deg,poly_exp.begin()+size,0);
    }
    void get_inv(){
        poly.resize(size()<<1);poly_inv.resize(size());rev.resize(size());vector<int>C(size());
        Poly_inv(poly,poly_inv,rev,C,num);
    }

    void get_ln(){
        get_inv();poly_ln.resize(size());
        Poly_ln(poly,poly_inv,poly_ln,rev,num,size());
    }

    void get_exp(){
        poly.resize(pg(num)<<1);rev.resize(size());poly_exp.resize(size());
        Poly_exp(poly,rev,poly_exp,num);
    }

    void operator*=(Poly &x){
        num=num+x.num-1;poly.resize(pg(num));rev.resize(size());x.poly.resize(size());
        for (int i = 0; i < size(); i++)rev[i]=(rev[i>>1]>>1)|((i&1)?(size()>>1):0);
        NTT(poly,rev,1);NTT(x.poly,rev,1);
        for (int i = 0; i < size(); i++)poly[i]=(ll)poly[i]*x[i]%mod;
        NTT(poly,rev,0);
    }


};


void ClearDewy(){

    for (int i = 0; i < n; i++)

        a[i]=read();

    }
    a.get_exp();
    for (int i = 0; i < n; i++)
    {
        writ(a.poly_exp[i],' ');
    }
}
